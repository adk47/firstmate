# Tier-2 endpoint allowlist for muso-admin-api.sh.
# One "METHOD /path" per line. Method is case-insensitive.
# Path matching: a param-less path matches as an EXACT string. A ":param"
# segment (e.g. /admin/user/:id) matches EXACTLY ONE path segment, so it gates
# "/admin/user/123" but NOT a sub-resource like "/admin/user/123/image".
# Lines starting with '#' and blank lines are ignored.
#
# Every endpoint listed here requires --confirmed-by <handle> matching
# tier2-approvers.txt AND --approval-msg-id <slack_ts> at call time.
# Dry-run (OPS_DRY_RUN=1) bypasses the gate.
#
# Add new entries deliberately. Removing an entry should require the
# same review as a tier-2 write itself.

POST /admin/profiles/merge
POST /admin/profiles/unmerge
POST /admin/tracks/merge
POST /admin/tracks/unmerge

# Destructive / account endpoints marked "⚠️ AARON ONLY" in
# muso-ops-admin/SKILL.md but previously enforced by DOCTRINE ONLY — they
# passed muso-admin-api.sh with just --requester. Now mechanically tier-2
# gated (require --confirmed-by + --approval-msg-id).
DELETE /admin/user/:id
DELETE /admin/track/:trackId
POST /admin/subscription
